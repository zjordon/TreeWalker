# 07 · 集成入口与示例

> 目标：把重放能力暴露给用户——编程 API、示例脚本、CLI flag、TUI 入口。

## 1. 编程 API

在 `Agent`（经 `rerun` mixin）上提供四个方法，对齐 browser-use 的 API 形态：

```python
class Agent(StepPipeline, RerunMixin):
    def save_history(self, file_path: str | Path | None = None) -> None: ...
    def detect_variables(self) -> dict[str, DetectedVariable]: ...
    async def rerun_history(self, history: AgentHistoryList, **kwargs) -> list[ActionResult]: ...
    async def load_and_rerun(
        self, history_file: str | Path,
        variables: dict[str, str] | None = None, **kwargs,
    ) -> list[ActionResult]: ...
```

**导出**：在 `src/tree_walker/__init__.py` 按需补 `DetectedVariable`、`StepMetadata`（`AgentHistoryList` 已导出）。

### 调用形态速查

```python
# 录制
settings.agent.max_steps = 10            # max_steps 在 settings 里设，不在 run() 里
agent = Agent(task=task, llm=llm, browser=browser, settings=settings.agent)
await agent.run()
agent.save_history("history.json")

# 检测可替换变量
variables = agent.detect_variables()    # {'email': DetectedVariable(...), ...}

# 重放（换数据）
agent2 = Agent(task="", llm=llm, browser=browser, settings=settings)
results = await agent2.load_and_rerun(
    "history.json",
    variables={"email": "jane@example.com"},
    max_step_interval=5,
    delay_between_actions=1,
    summary_llm=llm,
)
print(results[-1].success, results[-1].extracted_content)
```

## 2. 示例脚本

新建 `examples/features/rerun_history.py`（与其它 feature 示例同目录），仿 `examples/basic_agent.py` 风格（`load_settings` → `LLMClient` → `BrowserSession` → `Agent` → `run`），对齐 browser-use `rerun_history.py` 的三段式（录制 / 检测 / 重放）。

```python
"""Example: 录制 Agent 历史并换数据重放。

Prerequisites:
1. uv sync
2. chrome --remote-debugging-port=9222
3. 设置 ZHIPU_API_KEY
"""
import asyncio
import logging
import sys

sys.path.insert(0, f"{__file__}/../src")

from tree_walker import Agent, BrowserSession, LLMClient
from tree_walker.config import load_settings


async def main():
    settings = load_settings()
    if not settings.llm.api_key:
        print("Error: Set ZHIPU_API_KEY"); sys.exit(1)
    if not settings.browser.ws_url:
        print("Error: Chrome 未以 --remote-debugging-port=9222 启动"); sys.exit(1)

    llm = LLMClient(settings.llm)
    browser = BrowserSession(settings.browser)
    logging.basicConfig(level=logging.INFO)

    task = ("到 https://example.com/reference-number-form.html "
            "用示例数据填写表单、提交、提取 reference number")

    # ── 1. 录制 ──
    print("=== 录制 ===")
    settings.agent.max_steps = 10
    agent = Agent(task=task, llm=llm, browser=browser, settings=settings.agent)
    await agent.run()
    agent.save_history("agent_history.json")
    print("✓ 历史已保存到 agent_history.json")

    # ── 2. 检测变量 ──
    print("\n=== 检测变量 ===")
    variables = agent.detect_variables()
    for name, info in variables.items():
        fmt = f" (format: {info.format})" if info.format else ""
        print(f"  • {name}: {info.original_value!r}{fmt}")

    # ── 3. 换数据重放 ──
    if variables:
        new_values = {}
        if "email" in variables:      new_values["email"] = "jane.smith@example.com"
        if "full_name" in variables:  new_values["full_name"] = "Jane Smith"
        if "first_name" in variables: new_values["first_name"] = "Jane"
        if "date" in variables:       new_values["date"] = "1995-05-15"

        print("\n=== 重放（替换数据）===")
        replay_agent = Agent(task="", llm=llm, browser=browser, settings=settings.agent)
        results = await replay_agent.load_and_rerun(
            "agent_history.json",
            variables=new_values,
            max_step_interval=5,
            delay_between_actions=1,
            summary_llm=llm,
        )
        if results and results[-1].is_done:
            print(f"\n📊 摘要: {results[-1].extracted_content}")
            print(f"   成功: {results[-1].success}")


if __name__ == "__main__":
    asyncio.run(main())
```

## 3. CLI flag

`src/tree_walker/cli.py` 已用 click（`:15` `@click.command()`，现有 `--task`/`--debug`）。新增重放入口：

```python
@click.command()
@click.option("--task", "-t", default=None, help="初始任务")
@click.option("--debug", is_flag=True, default=False, help="调试模式")
@click.option("--rerun", "rerun_path", default=None, type=click.Path(exists=True),
              help="重放指定历史文件（不进入交互 TUI）")
@click.option("--var", "variables", multiple=True,
              help="变量替换 key=value（可重复，配合 --rerun）")
def main(task, debug, rerun_path, variables):
    settings = load_settings()
    settings.agent.enable_observability = True
    if debug:
        settings.tui.log_level = "DEBUG"

    if rerun_path:
        # 重放模式：不走 TUI，直接重放并打印摘要
        var_map = {}
        for kv in variables:
            k, _, v = kv.partition("=")
            var_map[k] = v
        asyncio.run(_rerun(settings, rerun_path, var_map))
    else:
        asyncio.run(_launch_tui(settings, task, debug))


async def _rerun(settings, rerun_path, variables):
    from tree_walker.browser.session import BrowserSession
    from tree_walker.llm.client import LLMClient
    llm = LLMClient(settings.llm)
    browser = BrowserSession(settings.browser)
    agent = Agent(task="", llm=llm, browser=browser, settings=settings.agent)
    results = await agent.load_and_rerun(
        rerun_path, variables=variables or None,
        max_step_interval=5, summary_llm=llm,
    )
    if results and results[-1].is_done:
        print(f"Success: {results[-1].success}\n{results[-1].extracted_content}")
```

**用法**：

```powershell
# 重放历史（无变量替换）
tw-tui --rerun agent_history.json

# 重放并替换变量
tw-tui --rerun agent_history.json --var email=jane@x.com --var full_name="Jane Smith"
```

## 4. TUI 入口（录制开关 + `/rerun` 命令）

`src/tree_walker/tui/app.py` 里两个入口（都已实现）：

### 4.1 录制开关（`Switch` widget）
- 输入区加 `Switch(value=False, id="record-switch")` +「录制」标签。`_is_recording()` 读其值。
- 开关**会话级**（默认关、不持久化）。仅**正常任务运行**录制；重放不录制。
- `_agent_worker` 里 `await agent.run(...)` 成功后调 `_maybe_save_recording()`：开关开 + 历史非空 →
  `agent.save_history(_record_filename())`，文件名 = 当前时间 `yyyyMMddhhmm.json`（24h），落 `rerun_history_dir`。
  ```python
  def _record_filename(now: datetime | None = None) -> str:
      return (now or datetime.now()).strftime("%Y%m%d%H%M") + ".json"
  ```
  > 同分钟两次录制会覆盖（分钟精度）。小时用 24h（`%H`），避免上下午歧义。

### 4.2 `/rerun` 命令（输入框拦截）
- TUI 无 `/command` 系统，故在 `on_multiline_input_submitted` 开头用模块函数 `_parse_rerun_command` 拦截：
  ```
  /rerun <相对路径> [key=value ...]
  ```
  解析后走 `_run_rerun(path, vars)`（仿 `_run_task`，但建 task="" 的 agent，跑 `load_and_rerun`）。
- 路径必须相对（走 `load_and_rerun` → `resolve_rerun_path` 的校验）；绝对路径 / `..` 越界在 `_rerun_worker` 抛 `ValueError` → 左栏提示。
- 重放事件经现有 `_setup_event_bridge`（EventBus）自动流到右栏——rerun 的 `_exec_one` 发 `ToolCallEvent`/`ToolResultEvent`，TUI 已订阅，无需特化。

### 用法
- 录制：打开「录制」开关 → 输任务回车 → 跑完看左栏「✓ 历史已保存：rerun-history/202607011430.json」。
- 重放：输入框敲 `/rerun 202607011430.json`（或带变量 `/rerun x.json email=a@b.com`）→ 回车。
